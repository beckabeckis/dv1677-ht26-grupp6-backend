# MongoDB API - CURL Test Commands
# 'handy' to copy/paste in the terminal, don't forgett to change the id!


# RESOURCES

# GET all resources
curl http://localhost:3000/api/resources/

# GET one resource
curl http://localhost:3000/api/resources/6ab1225a9bbb26f2f62b21d3

# POST new resource
curl -X POST http://localhost:3000/api/resources -H "Content-Type: application/json" -d '{"name":"Test","type":"Testtype","description":"Testing test","created_at":"2026-09-16 12:00",capacity":1}'

# PUT edit recource
curl -X PUT http://localhost:3000/api/resources/6ab1364f6ac593bc45c50e4b \
  -H "Content-Type: application/json" -d '{"id":"6ab1364f6ac593bc45c50e4b","name":"Test Changed","type":"Testtype","description":"Testing test","created_at":"2026-09-16 12:00","capacity":1}'

# DELETE recource
curl -X DELETE http://localhost:3000/api/resources/6ab1364f6ac593bc45c50e4b


# BOOKINGS

# GET all bookings
curl http://localhost:3000/api/bookings/

# GET all bookings of a resource
curl http://localhost:3000/api/bookings/resource/6ab13648dd7013fc09a2dbf2

# GET one booking
curl http://localhost:3000/api/bookings/6ab13648dd7013fc09a2dbf2

# POST new booking
curl -X POST http://localhost:3000/api/bookings -H "Content-Type: application/json" -d '{"resource_id":"6ab51de494195e6f16d651da","booked_by":"6ab51de494195e6f16d651d9","start_time":"2026-09-16 12:00","end_time":"2026-09-16 19:00","status":"confirmed"}'

# PUT edit booking
curl -X PUT http://localhost:3000/api/bookings/6ab1379c6ac593bc45c50e4e \
  -H "Content-Type: application/json" -d '{"id":"6ab1379c6ac593bc45c50e4e","resource_id":"6ab1369f6ac593bc45c50e4c","booked_by":"6ab51de494195e6f16d651d9","start_time":"2026-09-16 12:00","end_time":"2026-09-16 20:00","status":"confirmed"}'

# DELETE booking
curl -X DELETE http://localhost:3000/api/bookings/6ab1379c6ac593bc45c50e4e


# USERS

# GET all users
curl http://localhost:3000/api/users/

# GET one user
curl http://localhost:3000/api/users/6ab13648dd7013fc09a2dbf2

# POST new user
curl -X POST http://localhost:3000/api/users -H "Content-Type: application/json" -d '{"email":"test@student.bth.se","password_hash":"-","role":"student","created_at":"2026-09-16 12:00"}'

# PUT edit user
curl -X PUT http://localhost:3000/api/users/6ab1364f6ac593bc45c50e4b \
  -H "Content-Type: application/json" -d '{"id":"6ab1364f6ac593bc45c50e4b","email":"tested@student.bth.se","password_hash":"-","role":"student","created_at":"2026-09-16 12:00"}'

# DELETE user
curl -X DELETE http://localhost:3000/api/users/6ab1364f6ac593bc45c50e4b


# COMMENTS

# GET all comments
curl http://localhost:3000/api/comments/

# GET all comments of a booking
curl http://localhost:3000/api/comments/booking/6ab13648dd7013fc09a2dbf2

# GET all comments of a user
curl http://localhost:3000/api/comments/user/6ab13648dd7013fc09a2dbf2

# GET one comment
curl http://localhost:3000/api/comments/6ab13648dd7013fc09a2dbf2

# POST new comment
curl -X POST http://localhost:3000/api/comments -H "Content-Type: application/json" -d '{"booking_id":"6ab51de494195e6f16d651da","booked_by":"6ab51de494195e6f16d651d9","user_email":"erik@student.bth.se","text":"Testing","created_at":"2026-09-16 20:00"}'

# PUT edit comment
curl -X PUT http://localhost:3000/api/comments/6ab1379c6ac593bc45c50e4e \
  -H "Content-Type: application/json" -d '{"id":"6ab1379c6ac593bc45c50e4e","booking_id":"6ab51de494195e6f16d651da","booked_by":"6ab51de494195e6f16d651d9","user_email":"erik@student.bth.se","text":"Testing PUT","created_at":"2026-09-16 20:00"}'

# DELETE comment
curl -X DELETE http://localhost:3000/api/comments/6ab1379c6ac593bc45c50e4e
