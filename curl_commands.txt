# MongoDB API - CURL Test Commands
# 'handy' to copy/paste in the terminal, don't forgett to change the id!


# RESOURCES

# GET all resources
curl http://localhost:3000/api/resources/

# GET one resource
curl http://localhost:3000/api/resources/6ab1225a9bbb26f2f62b21d3

# POST new resource
curl -X POST http://localhost:3000/api/resources -H "Content-Type: application/json" -d '{"name":"Test","type":"Testtype","description":"Testing test","capacity":1}'

# PUT edit recource
curl -X PUT http://localhost:3000/api/resources/6ab1364f6ac593bc45c50e4b \
  -H "Content-Type: application/json" -d '{"id":"6ab1364f6ac593bc45c50e4b","name":"Test Changed","type":"Testtype","description":"Testing test","capacity":1}'

# DELETE recource
curl -X DELETE http://localhost:3000/api/resources/6ab1364f6ac593bc45c50e4b


# BOOKINGS

# GET all bookings
curl http://localhost:3000/api/bookings/

# GET one booking
curl http://localhost:3000/api/bookings/6ab13648dd7013fc09a2dbf2

# POST new booking
curl -X POST http://localhost:3000/api/bookings -H "Content-Type: application/json" -d '{"resource_id":"6ab1369f6ac593bc45c50e4c","user":"test@student.bth.se","start_time":"2026-09-16 12:00","end_time":"2026-09-16 19:00","status":"confirmed"}'

# PUT edit booking
curl -X PUT http://localhost:3000/api/bookings/6ab1379c6ac593bc45c50e4e \
  -H "Content-Type: application/json" -d '{"id":"6ab1379c6ac593bc45c50e4e","resource_id":"6ab1369f6ac593bc45c50e4c","user":"test@student.bth.se","start_time":"2026-09-16 12:00","end_time":"2026-09-16 20:00","status":"confirmed"}'

# DELETE booking
curl -X DELETE http://localhost:3000/api/bookings/6ab1379c6ac593bc45c50e4e
